  <threat id="T-09-10" severity="critical" stride="Tampering">
    Automatic staging/commit/tag/push could publish unrelated work or unsafe milestone claims.
    Mitigation: Plan is `autonomous: false`, writes approval package only, and requires explicit final user approval before any git publication action.
  </threat>
  <threat id="T-09-11" severity="high" stride="Information Disclosure">
    Staging plan could include secret-sensitive files or raw connection strings.
    Mitigation: include `secret_sensitive_excluded`, scan artifacts for `postgresql://`, and exclude `.env`/credential paths.
  </threat>
  <threat id="T-09-12" severity="medium" stride="Repudiation">
    Future maintainers may not know whether milestone was clean closed or accepted with blockers.
    Mitigation: `final_closure_decision.json`, `09-SUMMARY.md`, and `09-VERIFICATION.md` state closure route and accepted blockers explicitly.
  </threat>
</threat_model>

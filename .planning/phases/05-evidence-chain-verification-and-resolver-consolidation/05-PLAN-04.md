<threat_model>
  <threat id="T-05-08" severity="high" stride="Tampering">
    Metrics can be falsified by combining retrieval results from one corpus/version with judgments from another corpus/version.
    Mitigation: validate query_id/node_id/hit_count/corpus evidence before calculating metrics; fail closed on mismatch.
  </threat>
  <threat id="T-05-09" severity="medium" stride="Repudiation">
    A stale `judgment_completed.csv` can be reused accidentally and later mistaken for valid human judgment.
    Mitigation: integrity gate writes explicit pass/fail artifact with timestamp, retrieval file path, judgment file path, row counts, and mismatch reasons.
  </threat>
</threat_model>
